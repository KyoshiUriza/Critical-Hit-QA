Setup

Serve this project locally so the practice apps are reachable at http://localhost:8080:

# Python
python -m http.server 8080

# Node
npx serve -p 8080

# Then open http://localhost:8080

Update the BASE_URL in each example if you serve on a different port.

Login form — happy path + lockout

Target: practice-apps/login.html

// tests/login.spec.js
import { test, expect } from '@playwright/test';

const BASE_URL = 'http://localhost:8080';

test.describe('Login form', () => {
  test.beforeEach(async ({ page }) => {
    await page.goto(`${BASE_URL}/practice-apps/login.html`);
    await page.getByTestId('reset-lockout').click();
  });

  test('signs in with valid credentials', async ({ page }) => {
    await page.getByTestId('login-email').fill('demo@qa.test');
    await page.getByTestId('login-password').fill('Passw0rd!');
    await page.getByTestId('login-submit').click();
    await expect(page.getByTestId('login-result')).toContainText('Welcome');
  });

  test('shows validation errors on empty submit', async ({ page }) => {
    await page.getByTestId('login-submit').click();
    await expect(page.getByTestId('login-email-error')).toHaveText(/required/i);
    await expect(page.getByTestId('login-password-error')).toHaveText(/required/i);
  });

  test('locks account after 3 failed attempts', async ({ page }) => {
    for (let i = 0; i < 3; i++) {
      await page.getByTestId('login-email').fill('demo@qa.test');
      await page.getByTestId('login-password').fill('wrongPass1!');
      await page.getByTestId('login-submit').click();
    }
    await expect(page.getByTestId('login-result')).toContainText(/locked/i);
  });

  test('toggles password visibility', async ({ page }) => {
    const pw = page.getByTestId('login-password');
    await expect(pw).toHaveAttribute('type', 'password');
    await page.getByTestId('toggle-password').click();
    await expect(pw).toHaveAttribute('type', 'text');
  });
});

# test_login.py
import pytest
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

BASE_URL = "http://localhost:8080"

@pytest.fixture
def driver():
    d = webdriver.Chrome()
    yield d
    d.quit()

def testid(v):
    return (By.CSS_SELECTOR, f'[data-testid="{v}"]')

def test_valid_login(driver):
    driver.get(f"{BASE_URL}/practice-apps/login.html")
    driver.find_element(*testid("reset-lockout")).click()
    driver.find_element(*testid("login-email")).send_keys("demo@qa.test")
    driver.find_element(*testid("login-password")).send_keys("Passw0rd!")
    driver.find_element(*testid("login-submit")).click()
    result = WebDriverWait(driver, 5).until(
        EC.visibility_of_element_located(testid("login-result"))
    )
    assert "Welcome" in result.text

def test_lockout(driver):
    driver.get(f"{BASE_URL}/practice-apps/login.html")
    driver.find_element(*testid("reset-lockout")).click()
    for _ in range(3):
        driver.find_element(*testid("login-email")).clear()
        driver.find_element(*testid("login-email")).send_keys("demo@qa.test")
        driver.find_element(*testid("login-password")).clear()
        driver.find_element(*testid("login-password")).send_keys("wrongPass1!")
        driver.find_element(*testid("login-submit")).click()
    result = driver.find_element(*testid("login-result"))
    assert "locked" in result.text.lower()

Todo list — add, complete, filter

import { test, expect } from '@playwright/test';
const URL = 'http://localhost:8080/practice-apps/todo.html';

test.beforeEach(async ({ page }) => {
  await page.addInitScript(() => localStorage.clear());
  await page.goto(URL);
});

test('adds a todo and completes it', async ({ page }) => {
  await page.getByTestId('new-todo-input').fill('Write test plan');
  await page.getByTestId('add-todo').click();

  const item = page.locator('[data-testid=todo-item]').first();
  await expect(item.locator('[data-testid=todo-text]')).toHaveText('Write test plan');
  await expect(page.getByTestId('items-left')).toHaveText('1 item left');

  await item.locator('[data-testid=todo-toggle]').check();
  await expect(page.getByTestId('items-left')).toHaveText('0 items left');
});

test('does not add empty todos', async ({ page }) => {
  await page.getByTestId('new-todo-input').fill('   ');
  await page.getByTestId('add-todo').click();
  await expect(page.locator('[data-testid=todo-item]')).toHaveCount(0);
});

test('filters show correct items', async ({ page }) => {
  await page.getByTestId('new-todo-input').fill('one');
  await page.getByTestId('add-todo').click();
  await page.getByTestId('new-todo-input').fill('two');
  await page.getByTestId('add-todo').click();
  await page.locator('[data-testid=todo-item]').first().locator('[data-testid=todo-toggle]').check();

  await page.getByTestId('filter-active').click();
  await expect(page.locator('[data-testid=todo-item]')).toHaveCount(1);
  await page.getByTestId('filter-done').click();
  await expect(page.locator('[data-testid=todo-item]')).toHaveCount(1);
});
describe('Todo list', () => {
  beforeEach(() => {
    cy.clearLocalStorage();
    cy.visit('/practice-apps/todo.html');
  });

  it('adds and completes a todo', () => {
    cy.get('[data-testid=new-todo-input]').type('Write test plan{enter}');
    cy.get('[data-testid=todo-text]').first().should('have.text', 'Write test plan');
    cy.get('[data-testid=todo-toggle]').first().check();
    cy.get('[data-testid=items-left]').should('have.text', '0 items left');
  });
});

Shopping cart — totals, coupons

import { test, expect } from '@playwright/test';
const URL = 'http://localhost:8080/practice-apps/cart.html';

test('applies SAVE10 coupon and shows correct total', async ({ page }) => {
  await page.goto(URL);
  await page.getByTestId('add-widget').click();   // $10
  await page.getByTestId('add-widget').click();   // $10
  await page.getByTestId('add-gadget').click();   // $25 -> subtotal $45
  await expect(page.getByTestId('subtotal')).toHaveText('$45.00');

  await page.getByTestId('coupon-input').fill('save10');
  await page.getByTestId('apply-coupon').click();
  await expect(page.getByTestId('discount')).toHaveText('−$4.50');
  // subtotal $45 - $4.50 = $40.50; shipping $5.99 (under $50); tax 8% of $40.50 = $3.24
  // total = $40.50 + $5.99 + $3.24 = $49.73
  await expect(page.getByTestId('total')).toHaveText('$49.73');
});

test('rejects EXPIRED coupon', async ({ page }) => {
  await page.goto(URL);
  await page.getByTestId('add-gizmo').click();
  await page.getByTestId('coupon-input').fill('EXPIRED');
  await page.getByTestId('apply-coupon').click();
  await expect(page.getByTestId('coupon-msg')).toContainText(/expired/i);
});

API testing example (public sandbox)

Using https://jsonplaceholder.typicode.com as a public sandbox.

import { test, expect, request } from '@playwright/test';

test('GET /posts/1 returns expected shape', async () => {
  const api = await request.newContext();
  const res = await api.get('https://jsonplaceholder.typicode.com/posts/1');
  expect(res.status()).toBe(200);
  const body = await res.json();
  expect(body).toMatchObject({
    id: 1,
    userId: expect.any(Number),
    title: expect.any(String),
    body: expect.any(String),
  });
});

test('POST /posts creates a new resource', async () => {
  const api = await request.newContext();
  const res = await api.post('https://jsonplaceholder.typicode.com/posts', {
    data: { title: 'hello', body: 'world', userId: 1 },
  });
  expect(res.status()).toBe(201);
  const body = await res.json();
  expect(body.id).toBeDefined();
});
import requests

BASE = "https://jsonplaceholder.typicode.com"

def test_get_post():
    r = requests.get(f"{BASE}/posts/1")
    assert r.status_code == 200
    data = r.json()
    for key in ("id", "userId", "title", "body"):
        assert key in data

def test_post_creates_resource():
    r = requests.post(f"{BASE}/posts", json={"title": "hi", "body": "x", "userId": 1})
    assert r.status_code == 201
    assert "id" in r.json()

Bug-hunt regression suite (buggy builds)

Point the same automation at the buggy variants and each seeded defect becomes a failing test — the shape of a real regression suite catching real bugs.

// tests/login-broken.spec.js — each test asserts CORRECT behavior.
// Against practice-apps/login-broken.html they FAIL, surfacing seeded bugs.
import { test, expect } from '@playwright/test';
const URL = 'http://localhost:8080/practice-apps/login-broken.html';

test.beforeEach(async ({ page }) => {
  await page.addInitScript(() => { localStorage.clear(); sessionStorage.clear(); });
  await page.goto(URL);
});

test('BUG: email regex must require a TLD', async ({ page }) => {
  await page.getByTestId('login-email').fill('a@b');
  await page.getByTestId('login-password').fill('Passw0rd!');
  await page.getByTestId('login-submit').click();
  await expect(page.getByTestId('login-email-error')).toBeVisible();
});

test('BUG: email compare must be case-insensitive', async ({ page }) => {
  await page.getByTestId('login-email').fill('Demo@qa.test');
  await page.getByTestId('login-password').fill('Passw0rd!');
  await page.getByTestId('login-submit').click();
  await expect(page.getByTestId('login-result')).toContainText('Welcome');
});

test('BUG: empty password must be rejected', async ({ page }) => {
  await page.getByTestId('login-email').fill('demo@qa.test');
  await page.getByTestId('login-submit').click();
  await expect(page.getByTestId('login-password-error')).toBeVisible();
});

test('BUG: error must not distinguish "wrong password" from "no account"', async ({ page }) => {
  await page.getByTestId('login-email').fill('demo@qa.test');
  await page.getByTestId('login-password').fill('wrongPass1!');
  await page.getByTestId('login-submit').click();
  const knownMsg = await page.getByTestId('login-result').textContent();
  await page.getByTestId('reset-lockout').click();
  await page.getByTestId('login-email').fill('nobody@qa.test');
  await page.getByTestId('login-password').fill('wrongPass1!');
  await page.getByTestId('login-submit').click();
  const unknownMsg = await page.getByTestId('login-result').textContent();
  expect(knownMsg).toBe(unknownMsg); // Same message → no enumeration.
});

test('BUG: "Show" toggle must reverse to hidden', async ({ page }) => {
  const pw = page.getByTestId('login-password');
  await page.getByTestId('toggle-password').click();
  await expect(pw).toHaveAttribute('type', 'text');
  await page.getByTestId('toggle-password').click();
  await expect(pw).toHaveAttribute('type', 'password');
});

test('BUG: password must NOT be persisted', async ({ page }) => {
  await page.getByTestId('login-email').fill('demo@qa.test');
  await page.getByTestId('login-password').fill('Passw0rd!');
  await page.getByTestId('remember-me').check();
  await page.getByTestId('login-submit').click();
  const stored = await page.evaluate(() => ({ ...localStorage }));
  expect(stored).not.toHaveProperty('remembered_pw');
});

  await page.goto('http://localhost:8080/practice-apps/todo-broken.html');
  page.on('dialog', () => { throw new Error('XSS! alert() fired'); });
  await page.getByTestId('new-todo-input').fill('<img src=x onerror="alert(1)">');
  await page.getByTestId('add-todo').click();
  // Text should render literally.
  await expect(page.getByTestId('todo-text').first()).toContainText('<img');
});

Every one of these tests fails against the buggy build and passes against the clean build. That is exactly the pattern you want in a CI regression suite: green means the fix landed and stuck; red means someone broke a guarantee.
